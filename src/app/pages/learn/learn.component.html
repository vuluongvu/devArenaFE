insert(node, x):
    if node == null: return Node(x)
    if x < node.val:
        node.left = insert(node.left, x)
    else:
        node.right = insert(node.right, x)
    return node

search(node, x):
    if node == null or node.val == x:
        return node
    if x < node.val:
        return search(node.left, x)
    return search(node.right, x)

binarySearch(arr, target):
    left = 0, right = len - 1
    while left <= right:
        mid = (left + right) / 2
        if arr[mid] == target:
            return mid
        else if arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return -1

isPalindrome(s):
    left = 0, right = len - 1
    while left < right:
        if s[left] != s[right]:
            return false
        left++
        right--
    return true

maxSubarraySum(arr, k):
    windowSum = sum(arr[0..k-1])
    maxSum = windowSum
    for i from k to len - 1:
        windowSum += arr[i] - arr[i - k]
        maxSum = max(maxSum, windowSum)
    return maxSum
