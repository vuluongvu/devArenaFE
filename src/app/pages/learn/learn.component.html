#include <iostream>
#include <vector>

template <typename T>
class Stack {
private:
    std::vector<T> data;

public:
    // Thêm phần tử vào đỉnh - O(1)
    void push(const T& val) {
        data.push_back(val);
    }

    // Xóa phần tử ở đỉnh - O(1)
    void pop() {
        if (!empty()) {
            data.pop_back();
        }
    }

    // Xem phần tử ở đỉnh - O(1)
    T top() const {
        return data.back();
    }

    // Sửa phần tử ở đỉnh - O(1)
    void updateTop(const T& val) {
        if (!empty()) {
            data.back() = val;
        }
    }

    bool empty() const { return data.empty(); }
    size_t size() const { return data.size(); }
};
import java.util.ArrayDeque;
import java.util.Deque;

public class Stack<T> {
    private Deque<T> data = new ArrayDeque<>();

    // Thêm phần tử vào đỉnh - O(1)
    public void push(T val) {
        data.push(val);
    }

    // Xóa phần tử ở đỉnh - O(1)
    public T pop() {
        return data.pop();
    }

    // Xem phần tử ở đỉnh - O(1)
    public T peek() {
        return data.peek();
    }

    // Sửa phần tử ở đỉnh - O(1)
    public void updateTop(T val) {
        if (!data.isEmpty()) {
            data.pop();
            data.push(val);
        }
    }

    public boolean isEmpty() { return data.isEmpty(); }
    public int size() { return data.size(); }
}
class Stack:
    def __init__(self):
        self._data = []

    # Thêm phần tử vào đỉnh - O(1)
    def push(self, val):
        self._data.append(val)

    # Xóa phần tử ở đỉnh - O(1)
    def pop(self):
        if not self.is_empty():
            return self._data.pop()
        raise IndexError("pop from empty stack")

    # Xem phần tử ở đỉnh - O(1)
    def peek(self):
        if not self.is_empty():
            return self._data[-1]
        return None

    # Sửa phần tử ở đỉnh - O(1)
    def update_top(self, val):
        if not self.is_empty():
            self._data[-1] = val

    def is_empty(self):
        return len(self._data) == 0

    def size(self):
        return len(self._data)
