binarySearch(arr, target):
    left = 0, right = len - 1
    while left <= right:
        mid = (left + right) / 2
        if arr[mid] == target:
            return mid
        else if arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return -1

        left++
        right--
    return true

maxSubarraySum(arr, k):
    windowSum = sum(arr[0..k-1])
    maxSum = windowSum
    for i from k to len - 1:
        windowSum += arr[i] - arr[i - k]
        maxSum = max(maxSum, windowSum)
    return maxSum

quickSort(arr, low, high):
    if low < high:
        p = partition(arr, low, high)
        quickSort(arr, low, p - 1)
        quickSort(arr, p + 1, high)

partition(arr, low, high):
    pivot = arr[high], i = low
    for j from low to high - 1:
        if arr[j] < pivot:
            swap(arr[i++], arr[j])
    swap(arr[i], arr[high])
    return i
