Production Stack Pattern 
class Stack<T> {
  private items: T[] = [];

  push(item: T): void {
    this.items.push(item); // O(1)
  }

  pop(): T | undefined {
    return this.items.pop(); // O(1)
  }

  peek(): T | undefined {
    return this.items[this.items.length - 1]; // O(1)
  }
}
class Stack:
    def __init__(self):
        self._items = []

    def push(self, item):
        self._items.append(item)  # O(1)

    def pop(self):
        if not self.is_empty():
            return self._items.pop()  # O(1)

    def peek(self):
        if not self.is_empty():
            return self._items[-1]  # O(1)
