Stack Abstract Data Type (ADT) 
class Stack:
    constructor():
        items = []

    // Insert element at TOP — O(1)
    procedure push(value):
        items.append(value)

    // Delete and return element from TOP — O(1)
    procedure pop():
        if not isEmpty():
            return items.removeLast()
        raise UnderflowError

    // Read element at TOP without removal — O(1)
    procedure peek():
        if not isEmpty():
            return items.last()
        return null

    // Mutate value of element at TOP — O(1)
    procedure updateTop(value):
        if not isEmpty():
            items[lastIndex] = value

    function isEmpty():
        return length(items) == 0

    function size():
        return length(items)
